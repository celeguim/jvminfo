# enable metrics for minikube
minikube addons enable metrics-server

# Check if the metrics-server pod is running
kubectl get pods -n kube-system | grep metrics-server

# Test if you can see resource usage
kubectl top nodes
kubectl top pods

# create ns
kubectl create ns dev1

# deploy
kubectl apply -k kubernetes/kustomize/overlays/dev1/
kubectl apply -k kubernetes/kustomize/overlays/dev2/

# disable deployment
kubectl scale deployment.apps/dev1-jvminfo-v9-deployment -n dev1 --replicas=1
kubectl scale deployment.apps/dev2-jvminfo-v9-deployment -n dev2 --replicas=5

# port forward
kubectl port-forward -n dev1 deployment/dev1-jvminfo-v9-deployment 8081:8080 --address 0.0.0.0
kubectl port-forward -n dev2 deployment/dev2-jvminfo-v9-deployment 8082:8080 --address 0.0.0.0
kubectl port-forward -n dev1 service/dev1-jvminfo-v9-svc-ci 8083:30002 --address 0.0.0.0
